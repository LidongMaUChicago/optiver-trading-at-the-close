def reduce_mem_usage(df, verbose=True):
    start_mem = df.memory_usage().sum() / 1024**2
    for col in df.columns:
        col_type = df[col].dtype
        if col_type != object:
            c_min = df[col].min()
            c_max = df[col].max()
            if str(col_type)[:3] == 'int':
                if c_min > np.iinfo(np.int8).min and c_max < np.iinfo(np.int8).max:
                    df[col] = df[col].astype(np.int8)
                elif c_min > np.iinfo(np.int16).min and c_max < np.iinfo(np.int16).max:
                    df[col] = df[col].astype(np.int16)
                elif c_min > np.iinfo(np.int32).min and c_max < np.iinfo(np.int32).max:
                    df[col] = df[col].astype(np.int32)
            else:
                if c_min > np.finfo(np.float16).min and c_max < np.finfo(np.float16).max:
                    df[col] = df[col].astype(np.float16)
                elif c_min > np.finfo(np.float32).min and c_max < np.finfo(np.float32).max:
                    df[col] = df[col].astype(np.float32)
    end_mem = df.memory_usage().sum() / 1024**2
    if verbose:
        print('Memory usage before optimization is: {:.2f} MB'.format(start_mem))
        print('Memory usage after optimization is: {:.2f} MB'.format(end_mem))
        print('Decreased by {:.1f}%'.format(100 * (start_mem - end_mem) / start_mem))
    return df

def models(X_train, y_train, X_val, y_val):
    # Model training
    model = LinearRegression()  # You can adjust these parameters
    model.fit(X_train, y_train)
    
    # Coefficients (Feature Importance in context of Linear Regression)
    coefficients = model.coef_
    indices = np.argsort(np.abs(coefficients))[::-1]

    # Plotting feature importance (coefficients)
    plt.figure(figsize=(12, 6))
    plt.title("Feature Importances (Coefficients)")
    plt.bar(range(X_train.shape[1]), coefficients[indices], align="center")
    plt.xticks(range(X_train.shape[1]), [X_train.columns[i] for i in indices], rotation=90)
    plt.xlim([-1, X_train.shape[1]])
    plt.show()
    
    # Predict on training and validation set 
    y_train_pred = model.predict(X_train)
    y_val_pred = model.predict(X_val)
    
    # Calculate performance metrics
    train_r2 = r2_score(y_train, y_train_pred)
    train_mae = mean_absolute_error(y_train, y_train_pred)
    val_r2 = r2_score(y_val, y_val_pred)
    val_mae = mean_absolute_error(y_val, y_val_pred)

    print("Train R²:", train_r2, "Train MAE:", train_mae)
    print("Validation R²:", val_r2, "Validation MAE:", val_mae)
    
    # Statistical analysis using statsmodels
    X_train_sm = sm.add_constant(X_train)
    sm_model = sm.OLS(y_train, X_train_sm).fit()
    print(sm_model.summary())
    
    return model

# feature engineering

def feature_engineering(df):
    # Sort values
    df.sort_values(by = ['date_id', 'seconds_in_bucket'], inplace = True)
    
    # Original features
    features = ['imbalance_size', 'reference_price', 'matched_size', 
                'bid_price', 'bid_size', 'ask_price', 'ask_size', 'wap']

    # Calculate the bid-ask spread
    df['bid_ask_spread'] = df['ask_price'] - df['bid_price']

    # Change in WAP over time (momentum or shifts in market sentiment)
#     df['wap_change'] = df.groupby(['stock_id', 'date_id'])['wap'].diff()

    # Total quantity of bids and asks (liquidity and market depth)
    df['total_bid_size'] = df.groupby(['stock_id', 'date_id'])['bid_size'].transform('sum')
    df['total_ask_size'] = df.groupby(['stock_id', 'date_id'])['ask_size'].transform('sum')

    # Difference between bid and ask sizes (pressure on buy or sell side)
    df['order_imbalance'] = df['bid_size'] - df['ask_size']

    # Moving Averages Convergence Divergence (changes in trend strength, direction, momentum)
    df['macd'] = df.groupby(['stock_id', 'date_id'])['wap'].transform(lambda x: x.ewm(span=12, adjust=False).mean()) - \
                df.groupby(['stock_id', 'date_id'])['wap'].transform(lambda x: x.ewm(span=26, adjust=False).mean())

    # Rolling averages and volatility (5-minute window)
    df['bid_price_rolling_avg_5'] = df.groupby(['stock_id', 'date_id'])['bid_price'].transform(lambda x: x.rolling(window=5, min_periods=1).mean())
#     df['ask_price_volatility_5'] = df.groupby(['stock_id', 'date_id'])['ask_price'].transform(lambda x: x.rolling(window=5, min_periods=1).std())
    df['reference_price_ema_5'] = df.groupby(['stock_id', 'date_id'])['reference_price'].transform(lambda x: x.ewm(span=5, adjust=False).mean())

#     # Lag features for 'wap' (capturing past information)
#     num_lags = 3  # Number of lag values to consider
#     for lag in range(1, num_lags + 1):
#         df[f'wap_lag_{lag}'] = df.groupby(['stock_id', 'date_id'])['wap'].shift(lag)
    
    # Handling NaN values
    df.fillna(df.median(), inplace=True)

    # Add new features to the existing features list
    features += ['bid_ask_spread', 'total_bid_size', 'total_ask_size', 'order_imbalance', 'macd',
                'bid_price_rolling_avg_5', 'reference_price_ema_5'] 
#     features += ['bid_ask_spread', 'wap_change', 'total_bid_size', 'total_ask_size', 'order_imbalance', 'macd',
#                 'bid_price_rolling_avg_5', 'ask_price_volatility_5', 'reference_price_ema_5'] + \
#                 [f'wap_lag_{lag}' for lag in range(1, num_lags + 1)]

    # Statistics for each stock_id and date_id (mean and std deviation)
    price_columns = ['reference_price', 'bid_price', 'ask_price', 'wap']
    volume_columns = ['bid_size', 'ask_size']
    for col in price_columns + volume_columns:
#         stats = df.groupby(['stock_id', 'date_id'])[col].agg(['mean', 'std']).reset_index()
        stats = df.groupby(['stock_id', 'date_id'])[col].agg(['mean']).reset_index()
        stats.columns = [f"{col}_{stat}" if stat not in ['stock_id', 'date_id'] else stat for stat in stats.columns]
        df = df.merge(stats, on=['stock_id', 'date_id'], how='left')
    
#     # Add aggregated features to the features list
#     features += [f"{col}_{stat}" for col in ['reference_price', 'bid_price', 'ask_price', 'wap', 'bid_size', 'ask_size'] 
#                 for stat in ['mean', 'std']]

    # Add aggregated features to the features list
    features += [f"{col}_{stat}" for col in ['reference_price', 'bid_price', 'ask_price', 'wap', 'bid_size', 'ask_size'] 
                for stat in ['mean']]

    return df, features
