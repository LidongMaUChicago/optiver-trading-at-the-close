def reduce_mem_usage(df, verbose=True):
    start_mem = df.memory_usage().sum() / 1024**2
    for col in df.columns:
        col_type = df[col].dtype
        if col_type != object:
            c_min = df[col].min()
            c_max = df[col].max()
            if str(col_type)[:3] == 'int':
                if c_min > np.iinfo(np.int8).min and c_max < np.iinfo(np.int8).max:
                    df[col] = df[col].astype(np.int8)
                elif c_min > np.iinfo(np.int16).min and c_max < np.iinfo(np.int16).max:
                    df[col] = df[col].astype(np.int16)
                elif c_min > np.iinfo(np.int32).min and c_max < np.iinfo(np.int32).max:
                    df[col] = df[col].astype(np.int32)
            else:
                if c_min > np.finfo(np.float16).min and c_max < np.finfo(np.float16).max:
                    df[col] = df[col].astype(np.float16)
                elif c_min > np.finfo(np.float32).min and c_max < np.finfo(np.float32).max:
                    df[col] = df[col].astype(np.float32)
    end_mem = df.memory_usage().sum() / 1024**2
    if verbose:
        #print('Memory usage before optimization is: {:.2f} MB'.format(start_mem))
        #print('Memory usage after optimization is: {:.2f} MB'.format(end_mem))
        print('Decreased by {:.1f}%'.format(100 * (start_mem - end_mem) / start_mem))
    return df

# Use a simple neural network with an embedding layer to learn representations (embeddings) for each stock_id
# Train an Embedding Neural Network
def train_embedding_network(df, target_column=None, embedding_dim=2):
    """
    Trains an embedding neural network using stock_id and target from the provided DataFrame.
    
    Args:
    df (DataFrame): The DataFrame containing the stock_id and possibly the target.
    target_column (str, optional): The name of the target column in the DataFrame. 
                                   If None, the function assumes it's dealing with test data.
    embedding_dim (int): Dimension of the embedding vector.

    Returns:
    model (Sequential): Trained Keras Sequential model with embedding layer.
    num_stocks (int): Number of unique stocks.
    """
    num_stocks = len(df['stock_id'].unique())
    X_embedding = np.array(df['stock_id'])

    # For training data, prepare target; for test data, create a dummy target array
    y_embedding = np.array(df[target_column]) if target_column else np.zeros(len(df))

    model = Sequential()
    model.add(Embedding(input_dim=num_stocks + 1, output_dim=embedding_dim, input_length=1))
    model.add(Flatten())
    model.add(Dense(10, activation='relu'))
    
    if target_column:  # Only add output layer for training data
        model.add(Dense(1))  # Regression problem

    model.compile(optimizer='adam', loss='mean_squared_error')
    
    if target_column:  # Only fit the model for training data
        model.fit(X_embedding, y_embedding, epochs=3, batch_size=64)

    return model, num_stocks

def add_embeddings_to_dataframe(df, embedding_model):
    # Extract embeddings from the trained model
    embeddings = embedding_model.layers[0].get_weights()[0]

    # Create a DataFrame for the embeddings
    embedding_cols = [f'embedding_{i}' for i in range(embeddings.shape[1])]
    embedding_df = pd.DataFrame(embeddings, columns=embedding_cols)

    # Create a 'stock_id' column in embedding_df
    embedding_df['stock_id'] = range(embeddings.shape[0])

    # Merge the original DataFrame with the embedding DataFrame
    merged_df = df.merge(embedding_df, on='stock_id', how='left')

    return merged_df

# train the embedding model

# feature engineering

def feature_engineering(df):
    # Original features
    ori_features = ['imbalance_size', 'imbalance_buy_sell_flag', 'reference_price', 'matched_size', 'far_price', 'near_price',
                    'bid_price', 'bid_size', 'ask_price', 'ask_size', 'wap', 
                    'embedding_0', 'embedding_1']
    # Instantaneous Features
    df['bid_ask_spread']  = df['ask_price'] - df['bid_price']
    df['ref_bid_spread']  = df['reference_price'] - df['bid_price']
    df['order_imbalance'] = df['bid_size'] - df['ask_size']
    df['market_depth']    = df['bid_size'] + df['ask_size']
    df['vol_imb_ratio']   = df['order_imbalance'] / df['market_depth']
    df['mktcap']          = df['matched_size'] * df['reference_price']
    df['far_price']       = df['far_price'].fillna(0)
    df['near_price']      = df['near_price'].fillna(0)
    df['phase1']          = np.where(df['seconds_in_bucket']<300, 1, 0)
    df['phase2']          = np.where((df['seconds_in_bucket']>=300) & (df['seconds_in_bucket']<480), 1, 0)
    df['phase3']          = np.where(df['seconds_in_bucket']>=480, 1, 0)
    df['switch']          = np.where(((df['seconds_in_bucket'] >= 240) & (df['seconds_in_bucket'] < 300)) |
                                     ((df['seconds_in_bucket'] >= 420) & (df['seconds_in_bucket'] < 480)), 1, 0)
    df['sec_in_phrase']   = np.select([df['phase1']==1, df['phase2']==1, df['phase3']==1], 
                                      [df['seconds_in_bucket'], df['seconds_in_bucket']-300, df['seconds_in_bucket']-480], 
                                      default=np.nan)
    
    # Update features list
    ori_features += ['bid_ask_spread', 'ref_bid_spread', 'market_depth', 'vol_imb_ratio', 'mktcap',  
                     'phase1', 'phase2', 'phase3', 'switch', 'sec_in_phrase']
    
    ######################### Nearest Neighbors #####################
    # lag setting
    n_lags = 2 
    lag_columns = ['imbalance_size', 'imbalance_buy_sell_flag', 'reference_price', 'matched_size', 'wap',
                   'far_price', 'near_price', 'bid_ask_spread', 'vol_imb_ratio']
    
    # rolling setting
    window_size = 3
    rol_columns = ['imbalance_size', 'reference_price', 'wap', 'bid_ask_spread', 'mktcap']
    
    ######################################## Stock Neighbor ################################################
    df['time_id'] = df['time_id'].astype('float32')
    df.sort_values(by=['time_id', 'stock_id'], inplace=True)
    
    sn_rol_dfs = []
    for col in rol_columns: 
        sn_rolavg_col  = df.groupby(['time_id'])[col].rolling(window=3, min_periods=1).mean().reset_index(level=[0, 1], drop=True)
        sn_rol_dfs.append(sn_rolavg_col.rename(f'{col}_rolavg_nei'))
        ori_features.append(f'{col}_rolavg_nei')
    
    update_df = pd.concat([df] + sn_rol_dfs, axis=1)
    df        = reduce_mem_usage(update_df)
    
    
    ######################################## Time Series ################################################
    df['time_id'] = df['time_id'].astype('float32')
    df.sort_values(by=['date_id', 'stock_id', 'seconds_in_bucket'], inplace=True)
    
    ts_lag_dfs = []
    for col in lag_columns:
        for lag in range(1, n_lags + 1):
            ts_lag_col = df.groupby(['stock_id', 'date_id'])[col].shift(lag)
            ts_lag_col.fillna(df[col],inplace=True)
            ts_lag_dfs.append(ts_lag_col.rename(f'{col}_lag_{lag}'))
            if lag==1:
                ori_features.append(f'{col}_lag_{lag}')
            
    update_df = pd.concat([df] + ts_lag_dfs, axis=1)
    df        = reduce_mem_usage(update_df)

    # time-series features
    df['wap_return']       = (df['wap'] - df['wap_lag_1']) / df['wap_lag_1']
    df['wap_return_lag_1'] = df.groupby(['stock_id', 'date_id'])['wap_return'].shift(1)
    
    df['imb_chg']          = df['imbalance_size'] * df['imbalance_buy_sell_flag'] - df['imbalance_size_lag_1'] * df['imbalance_buy_sell_flag_lag_1']
    imb_conditions = [
        df['imbalance_size'] == df['imbalance_size_lag_1'], 
        (df['imbalance_size'] != df['imbalance_size_lag_1']) & (df['imbalance_size_lag_1'] == 0), 
        (df['imbalance_size'] != df['imbalance_size_lag_1']) & (df['imbalance_size_lag_1'] != 0) 
    ]
    imb_values = [
        0,
        1,
        df['imb_chg']/df['imbalance_size_lag_1']
    ]
    df['imb_chg_rate']       = np.select(imb_conditions, imb_values, default=np.nan)
    df['imb_chg_rate_lag_1'] = df.groupby(['stock_id', 'date_id'])['imb_chg_rate'].shift(1)
    
    df['matched_chg']    = df['matched_size'] - df['matched_size_lag_1']
    df['filled_imb']     = df['imb_chg'] + df['matched_chg']
    df.fillna(0, inplace=True)
                                     
    ori_features += ['wap_return', 'wap_return_lag_1', 'imb_chg', 'imb_chg_rate', 'imb_chg_rate_lag_1', 'matched_chg', 'filled_imb']
    
    # liquidity rolling average
    ts_rol_dfs = []
    for col in rol_columns: 
        ts_rolavg_col  = df.groupby(['stock_id', 'date_id'])[col].rolling(window=3, min_periods=1).mean().reset_index(level=[0, 1], drop=True)
        ts_rol_dfs.append(ts_rolavg_col.rename(f'{col}_rolavg'))
        ori_features.append(f'{col}_rolavg')
    
    update_df = pd.concat([df] + ts_rol_dfs, axis=1)
    df        = reduce_mem_usage(update_df)
    
    df.fillna(0, inplace=True)
    df['time_id'] = df['time_id'].astype('float32')
    df.sort_values(by=['time_id', 'stock_id'], inplace=True)

    return df, ori_features

def train_xgboost_model(X_train, y_train, X_val=None, y_val=None):
   
    # Initialize and train the XGBoost model
    model = XGBRegressor(n_estimators=200, max_depth=3, learning_rate=0.1, random_state=42)
    
    # Use a simpler XGBoost model setup 
#     model = XGBRegressor(n_estimators=50, max_depth=3, learning_rate=0.1, random_state=42)

    # If validation data is provided, use early stopping
    if X_val is not None and y_val is not None:
        model.fit(X_train, y_train, eval_set=[(X_val, y_val)], verbose=False)
    else:
        model.fit(X_train, y_train, verbose=False)

    # Predict on training set
    y_train_pred = model.predict(X_train)

    # Calculate MAE and R² metrics for the training set
    train_mae = mean_absolute_error(y_train, y_train_pred)
    train_r2 = r2_score(y_train, y_train_pred)
    print("Train MAE:", train_mae, "Train R²:", train_r2)
    
    # If validation data is provided, evaluate model on it
    if X_val is not None and y_val is not None:
        y_val_pred = model.predict(X_val)
        val_mae = mean_absolute_error(y_val, y_val_pred)
        val_r2 = r2_score(y_val, y_val_pred)
        print("Validation MAE:", val_mae, "Validation R²:", val_r2)

    # Feature Importance Analysis
    print("\nFeature Importances:")
    feature_importances = model.feature_importances_
    for feature_name, importance in zip(X_train.columns, feature_importances):
        print(f"{feature_name}: {importance}")

    return model

def xgb_cross_validation(df, features, n_splits=5):
    total_mae_score = 0
    total_r2_score = 0

    # Determine the range of date_id and split it
    date_ids = df['date_id'].unique()
    split_size = len(date_ids) // n_splits

    for i in range(n_splits):
        # Define the split for the current fold
        val_start = i * split_size
        val_end = val_start + split_size
        val_date_ids = date_ids[val_start:val_end]

        # Split the data
        train_split = df[~df['date_id'].isin(val_date_ids)]
        val_split = df[df['date_id'].isin(val_date_ids)]

        X_train = train_split[features]
        y_train = train_split['target']
        X_val = val_split[features]
        y_val = val_split['target']

        # Train XGBoost model
        model = XGBRegressor(n_estimators=200, learning_rate=0.1, max_depth=3, random_state=42)
        
        # Use a simpler XGBoost model setup inside the loop
#         model = XGBRegressor(n_estimators=50, learning_rate=0.1, max_depth=3, random_state=42)
    
        model.fit(X_train, y_train, eval_set=[(X_val, y_val)], verbose=False)

        # Evaluate the model
        y_val_pred = model.predict(X_val)
        mae = mean_absolute_error(y_val, y_val_pred)
        r2 = r2_score(y_val, y_val_pred)
        print(f"Fold {i+1}, Validation MAE: {mae}, Validation R²: {r2}")
        total_mae_score += mae
        total_r2_score += r2

    average_mae = total_mae_score / n_splits
    average_r2 = total_r2_score / n_splits
    print(f"Average MAE over {n_splits} folds: {average_mae}")
    print(f"Average R² over {n_splits} folds: {average_r2}")

def prediction(test, model, features, buffer_df):
    # Apply embeddings to test data
    test = add_embeddings_to_dataframe(test, embedding_model)
    
    # Combine the buffer with the new chunk of test data
    combined_data = pd.concat([buffer_df, test], ignore_index=True)

    # Apply feature engineering to the combined data
    combined_feature, _ = feature_engineering(combined_data)

    # Fill any remaining NaN values after feature engineering
    combined_feature.fillna(combined_feature.median(), inplace=True)

    # Extract the feature matrix for the test data
    X_test = combined_feature[features][-len(test):]

    # Make predictions using the XGBoost model
    y_predict = model.predict(X_test)

    # Update the buffer with the last 'buffer_size' rows of the combined data
    new_buffer_df = combined_data[-buffer_size:]

    return y_predict, new_buffer_df, X_test
